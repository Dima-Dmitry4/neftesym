12  Разбор 3. Купол с разломами: блоки и поддержка

neft = {
  const m = await import(new URL("assets/js/neft.js", document.baseURI).href);
  await m.ready();
  return m;
}
core = neft.core()

Учебный купол (сценарий base, тот же, что в главе 4) разрезан двумя разломами: F-1 на востоке (по замыслу – запечатан, сброс 4 м) и F-2 на западе (пропускает 30 % потока). Нефтяная зона – в основном в центральном блоке, но восточный блок тоже содержит нефть. Вопрос разбора: нужна ли отдельная нагнетательная за разломом и как понять это по ходу игры, если герметичность разлома скрыта.

12.1 Опыт

Три добывающие: Д-1 – в восточном блоке за F-1, Д-2 и Д-3 – в центральном блоке. Нагнетательная Н-1 – в центральном блоке. Второй вариант – добавлена Н-2 в восточном блоке. Расчёт – 5 лет.

Переключатель «Запечатан / со щелями» связан с находкой главы 4: до исправления у «запечатанного» F-1 в ядре игры были щели, и он работал как частично проницаемый. Сейчас игра считает разлом запечатанным по-настоящему.

viewof gaps13 = Inputs.radio(new Map([["запечатан (как в игре)", false], ["со щелями (ядро до исправления)", true]]), { label: "F-1", value: false })
viewof go13 = Inputs.button("▶ Рассчитать варианты")
Код вариантов (ядро игры) 
variants13 = {
  const base = (east) => () => {
    const f = neft.field({ scenario: "base", strictGaps: gaps13 });
    f.drill({ x: 1300, y: 1150 });                       // Д-1: восточный блок
    f.drill({ x: 900, y: 1200 }); f.drill({ x: 900, y: 750 });
    f.drill({ x: 700, y: 950, type: "inj" });            // Н-1: центральный блок
    if (east) f.drill({ x: 1300, y: 800, type: "inj" }); // Н-2: восточный блок
    return f;
  };
  return [["Только Н-1", base(false)], ["Н-1 + Н-2 за разломом", base(true)]];
}
run13 = {
  if (!go13) { yield null; return; }
  for await (const st of neft.compare(variants13, 5, { step: 60 })) yield st;
}
{
  const el = neft.report(run13, { layer: "P", charts: ["oil_td", "avgP", "money", "kin"], mapWidth: 340 });
  if (run13 && run13.done) {
    const t = run13.variants.map((v) => { const w = v.f.well("Д-1"); return `<tr><td>${v.name}</td><td>${w.q.toFixed(0)}</td><td>${v.f.sim.P[w.cell].toFixed(0)}</td><td>${(w.cum.oil * 0.86 / 1000).toFixed(0)}</td></tr>`; }).join("");
    el.prepend(Object.assign(document.createElement("div"), { innerHTML: `<p><b>Д-1 за разломом через 5 лет</b></p><table class="table table-sm" style="max-width:560px"><thead><tr><th>Вариант</th><th>дебит, м³/сут</th><th>давление в ячейке, атм</th><th>нефть, тыс. т</th></tr></thead><tbody>${t}</tbody></table>` }));
  }
  return el;
}

12.2 Разбор

  • За запечатанным разломом Д-1 без своей нагнетательной за пять лет полностью истощает блок: давление падает до 20 атм (минимум на приёме ЭЦН), дебит – до нуля. Н-2 возвращает скважину к жизни и окупается многократно: +230 млн руб ЧДД при цене скважины 80 млн.
  • Даже при щелях в разломе (так считала игра до исправления) Н-2 даёт +120 млн руб: поддержка через щели и обход слабее, чем прямая закачка.
  • Карта давления показывает разлом как ступеньку изобар. В игре карта давления скрыта до конца партии – значит, нужно узнавать о разломе косвенно.

12.3 Как распознать запечатанный разлом в игре

  1. Сейсмика показывает разлом и сброс, но не герметичность.
  2. Динамика: у скважины за разломом дебит и забойное давление падают быстрее, чем у соседей, хотя обводнённость не растёт (вода от нагнетательных не доходит).
  3. ГДИС: пластовое давление по КВД у скважины за разломом заметно ниже, чем у скважин в блоке с закачкой. Поздняя часть КВД «видит» разлом как непроницаемую границу – производная поднимается.

Проведём ГДИС на двух скважинах после двух лет работы:

const f = neft.field({ scenario: "base" });
const d1 = f.drill({ x: 1300, y: 1150 }), d2 = f.drill({ x: 900, y: 1200 });
f.drill({ x: 900, y: 750 }); f.drill({ x: 700, y: 950, type: "inj" });
await f.runAll(2, { step: 60 });
for (const w of [d1, d2]) {
  const bu = core.pressureBuildup(f.sim, w, 30), p = core.buildupPres(bu);
  print(`${w.name}: дебит до остановки ${w.q.toFixed(0)} м³/сут, Pwf ${w.pwf.toFixed(0)} атм → пластовое по КВД ${p.pRes ? p.pRes.toFixed(0) : "≥ " + p.pEnd.toFixed(0)} атм (${p.method})`);
}
СоветПравило

Пока не доказано обратное, считайте каждый блок за разломом отдельной залежью со своей системой ППД. Одна лишняя нагнетательная стоит 80 млн руб, а истощённый блок – потерянные запасы и простаивающие добывающие.
