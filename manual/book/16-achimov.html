15  Разбор 6. Ачимовка: трудноизвлекаемые запасы

neft = {
  const m = await import(new URL("assets/js/neft.js", document.baseURI).href);
  await m.ready();
  return m;
}
core = neft.core()

Ачимовская толща (Среднее Приобье) – турбидитные песчаники и алевролиты, проницаемость 0,01–0,5 мД (медиана 0,07 мД), аномально высокое пластовое давление (≈ 324 атм на 2800 м), дорогая скважина (95 млн руб). Сложность ★★★★. Это трудноизвлекаемые запасы (ТрИЗ): нефти много, но пласт её почти не отдаёт. Эталон здесь даёт КИН 26 % и ЧДД ≈ 1,5 млрд руб – самый низкий из всех объектов. Уроки: заканчивание решает всё и закачку надо начинать вовремя.

15.1 Опыт 1. Заканчивание

Одна добывающая в центральном блоке на ЭЦН-125 и вертикальная нагнетательная, 3 года:

viewof go16a = Inputs.button("▶ Рассчитать заканчивания")
Код вариантов (ядро игры) 
variants16a = {
  const mk = (opt) => () => {
    const f = neft.field({ scenario: "ach" });
    f.drill({ x: 1050, y: 700, pump: "125", ...opt });
    f.drill({ x: 1250, y: 1050, type: "inj", choke: 12 });
    return f;
  };
  return [["ВС", mk({ kind: "v" })], ["ВС + ГРП 120 м", mk({ kind: "frac", xf: 120, az: 90 })], ["ГС 600 м", mk({ kind: "hs", L: 600, az: 0 })], ["ГС 600 м + МГРП 5×80", mk({ kind: "mgrp", L: 600, az: 0, n: 5, xf: 80 })]];
}
run16a = {
  if (!go16a) { yield null; return; }
  for await (const st of neft.compare(variants16a, 3, { step: 60 })) yield st;
}
neft.report(run16a, { layer: "P", charts: ["oil_td", "money"], mapWidth: 260 })
Заканчивание Дебит жидкости к 3-му году, м³/сут Нефть за 3 года, тыс. т
ВС 2 1,7
ВС + ГРП 120 м 8 11,6
ГС 600 м 9 10,6
ГС 600 м + МГРП 5×80 15 27,9

Вертикальная скважина в пласте 0,07 мД – 2 м³/сут: это даже не окупает обслуживание. ГРП и ГС дают в 5–7 раз больше, МГРП – в 16 раз. Здесь F_{CD}=1000/(0{,}07\cdot80)\approx180: трещина почти бесконечной проводимости, и каждая стадия работает в полную силу. За три года все варианты ещё в минусе – ТрИЗ окупаются долго.

15.2 Опыт 2. Когда начинать закачку

Две ГС 900 м с МГРП 8×40 (как в эталоне) на расстоянии 300 м, 12 лет:

viewof go16b = Inputs.button("▶ Рассчитать стратегии (12 лет)")
Код вариантов (ядро игры) 
variants16b = {
  const hs = (f, x, type = "prod") => f.drill({ x, y: 550, kind: "mgrp", L: 900, az: 0, n: 8, xf: 40, type, pump: "125", choke: 26 });
  const F = () => neft.field({ scenario: "ach", pKNS: 130 });
  return [
    ["Закачка с первого дня", () => { const f = F(); hs(f, 1000); hs(f, 1300, "inj"); return f; }],
    ["Год на нефти, потом закачка", () => {
      const f = F(); hs(f, 1000); const w = hs(f, 1300);
      f.onYear = (yr) => { if (yr === 1) f.convert(w, { choke: 26 }); };
      return f; }],
    ["Без закачки", () => { const f = F(); hs(f, 1000); hs(f, 1300); return f; }],
  ];
}
run16b = {
  if (!go16b) { yield null; return; }
  // compare с действием в конце первого года: перевод второй ГС под закачку
  const V = variants16b.map(([name, build]) => ({ name, build }));
  const total = V.length * 12 * 365; let done = 0, last = 0;
  for (const v of V) {
    v.f = v.build(); v.rows = v.f.rows;
    for (let yr = 1; yr <= 12; yr++) {
      for await (const row of v.f.run(1, { step: 73 })) {
        done += 73;
        if (performance.now() - last > 250) { last = performance.now(); yield { done: false, frac: done / total, text: `«${v.name}»: год ${row.year.toFixed(1)} из 12`, variants: V }; }
      }
      if (v.f.onYear) v.f.onYear(yr);
    }
  }
  yield { done: true, frac: 1, variants: V };
}
neft.report(run16b, { layer: "P", charts: ["oil_td", "avgP", "kin", "money"], mapWidth: 300 })
Стратегия Нефть за 12 лет, тыс. т КИН ЧДД, млн руб Дебит добывающей к 12-му году, м³/сут
Закачка с первого дня 80 4,0 % 65 22
Год на нефти, потом закачка 94 4,7 % 201 22
Без закачки 88 4,5 % 204 3–9

15.3 Разбор

  • Год на нефти выгоднее закачки с первого дня: вторая скважина за год добывает нефть и окупает часть своей стоимости, а пластовое давление первые годы и так высокое (АВПД). Эталон ачимовки делает именно так: все будущие нагнетательные первый год работают на нефть.
  • Без закачки ЧДД за 12 лет такой же (204 против 201 млн руб), а нефти меньше – и дебиты к концу в 3–4 раза ниже, чем с закачкой, а давление снизилось с 324 до ≈ 235 атм. В отчёте игры оценка идёт за 20 лет (10 факта + 10 прогноза), и вариант без ППД в прогнозе «доживает» на минимальных дебитах, а вариант с закачкой держит 22 м³/сут. КИН – половина оценки.
  • Небольшой КИН – не провал. 4–5 % за 12 лет на двух скважинах – нормально для ТрИЗ; эталон получает 26 % за 20 лет семью скважинами (3 добывающие ГС + МГРП 8×40 и 4 нагнетательные) на сетке 280 м.
СоветРецепт для ачимовки

ГС 900–1000 м с МГРП 8 стадий по 40 м (короткие трещины дешевле, а при F_{CD}\gg1 длина трещины мало что даёт) → плотная шахматная сетка 280 м → будущие нагнетательные первый год на нефти, затем перевод → ЭЦН малых типоразмеров (дебиты 15–30 м³/сут). Сейсмика обязательна, но съедает большую долю ЧДД – ошибаться с расстановкой нельзя.

15.4 Вопросы

  1. Почему МГРП 4 стадии иногда дают больший ЧДД, чем 8 стадий? Проверьте в живой ячейке ниже.
  2. Через сколько лет окупается одна ГС + МГРП на ачимовке?
  3. Почему эталон на ачимовке выбрал 8 стадий МГРП, а на АВ1 – 5, хотя трещины у обоих короткие (40 м)?
const res = [];
for (const n of [2, 4, 8, 11]) {
  const f = neft.field({ scenario: "ach" });
  f.drill({ x: 1000, y: 550, kind: "mgrp", L: 900, az: 0, n, xf: 40, pump: "125" });
  await f.runAll(6, { step: 73 });
  const s = f.stats();
  res.push({ "стадий": n, "нефть, тыс. т": (s.cumOil_t / 1e3).toFixed(1), "ЧДД за 6 лет, млн руб": s.npv.toFixed(0) });
}
return res;
