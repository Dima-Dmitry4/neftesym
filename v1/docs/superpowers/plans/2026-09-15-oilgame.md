# Oilgame Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Учебная игра-симулятор разработки нефтяного месторождения в одном HTML-файле (IMPES: давление + насыщенность, скважины по Писману, экономика в рублях).

**Architecture:** Один `index.html` с двумя скриптами: `#sim-core` (чистое ядро без DOM: сетка, IMPES, скважины, экономика, самотест) и `#app` (UI: рендер, графики, экраны). Тесты — node-харнесс `tests/selftest.js`, извлекающий `#sim-core` из HTML и исполняющий его в `vm`-песочнице; те же проверки доступны в браузере кнопкой «Самотест».

**Tech Stack:** Vanilla JS (ES2020), Canvas 2D, без зависимостей; тесты на node ≥ 18 (файл `tests/selftest.js`, запуск `node tests/selftest.js`).

## Global Constraints (из спека docs/superpowers/specs/2026-09-15-oilgame-design.md)

```js
const GRID = { nx: 50, ny: 50, dx: 40, h: 10, phi: 0.2 };
const DEFAULT_PARAMS = {
  mu_o: 5, price_oil: 10000, drill_cost: 10e6, lift_cost: 500, inj_cost: 300,
  horizon_days: 3650, init_money: 50e6,
};
function mulberry32(seed) {
  let a = seed >>> 0;
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0);
  };
}
function gauss(rng) { // Box–Muller
  const u = (rng() + 1) / 4294967297 || 1e-12, v = rng() / 4294967296;
  return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * v);
}
function buildReservoir(params) {
  const { nx, ny, dx, h, phi } = GRID;
  const rng = mulberry32(20260915);
  const cx = (nx - 1) / 2, cy = (ny - 1) / 2;
  const active = new Uint8Array(nx * ny), k = new Float64Array(nx * ny);
  // русло: синусоидальный канал через поле (центральная линия с изгибом)
  const river = (i) => cy + 6 * Math.sin(i / nx * 2 * Math.PI) + 3 * Math.sin(i / 7.0);
  for (let j = 0; j < ny; j++) {
    for (let i = 0; i < nx; i++) {
      const c = j * nx + i;
      // контур: эллипс с гармоническими возмущениями
      const th = Math.atan2((j - cy) / 1.0, (i - cx) / 1.25);
      const wob = 1 + 0.10 * Math.sin(3 * th + 0.7) + 0.06 * Math.sin(5 * th - 1.3);
      const ri = Math.hypot((i - cx) / (23 * 1.25), (j - cy) / 23);
      if (ri / wob < 1) {
        active[c] = 1;
        const kbg = 20 * Math.exp(gauss(rng) * 0.7);          // фон 20–150 мД (логнорм.)
        const inRiver = Math.abs(j - river(i)) < 2.6;
        const kch = 300 + 500 * (rng() / 4294967296);          // канал 300–800 мД
        k[c] = inRiver ? Math.max(kch, kbg) : kbg;
      }
    }
  }
  const poreVol = new Float64Array(nx * ny);
  const cells = [];
  for (let c = 0; c < nx * ny; c++)
    if (active[c]) { poreVol[c] = dx * dx * h * phi; cells.push(c); }
  const ooipM3 = cells.reduce((s, c) => s + poreVol[c] * (1 - 0.2), 0);
  return { nx, ny, dx, h, phi, ct: 1e-4, P0: 250, mu_o: params.mu_o, mu_w: 1,
           Swc: 0.2, Sor: 0.25, rho_oil: 0.86, active, k, poreVol, ooipM3, cells };
}
```

```js
const CONV_K = 0.00864; // м³/сут · сП / (мД · м · атм)
function krw(Sw, r) { const s = (Sw - r.Swc) / (1 - r.Swc - r.Sor); return Math.max(0, Math.min(1, s)); }
function kro(Sw, r) { const s = (1 - r.Sor - Sw) / (1 - r.Swc - r.Sor); return Math.max(0, Math.min(1, s)); }
function fw(Sw, r) { const a = krw(Sw, r) / r.mu_w, b = kro(Sw, r) / r.mu_o; return (a + b) > 0 ? a / (a + b) : 0; }
function lambdaT(Sw, r) { return krw(Sw, r) / r.mu_w + kro(Sw, r) / r.mu_o; }
function wellIndex(k, r) { return CONV_K * 2 * Math.PI * k * r.h / Math.log(8 / 0.1); }
```

### Task 4: Решатель давления (матрично-свободный CG) и состояние симуляции

**Files:** Modify: `index.html` (`#sim-core`)

**Interfaces:**
- Produces:
  - `createSim(reservoir, params)` → sim-объект: `{ r, P:Float64Array, Sw:Float64Array, wells:[], t:0, dayCounter:0, history:{...} }`
    + методы `addWell(cellIdx, type /*"prod"|"inj"*/, pwf)`, `removeWell` (не нужен — YAGNI), `solvePressure(dt)`, `computeFaceFluxes() → Float64Array flux[4·?]` (внутреннее), `transportSubstep(dt)`, `cflLimit() → dt`, `advance(days)`, `avgPressure()`, `stats()`.
  - Лицевые проводимости считаются на лету: `faceT(c, nb, dir)` = CONV_K·k_harm·h·(dy/dx=1)·λt(upstream по знаку потока с прошлых потоков; на первом шаге — среднее арифметическое λ соседей). Для матричной симметрии CG берём λ_up по дрейфующему направлению — допускается небольшая несимметрия; если CG не сходится — λ арифметическая. Решение: использовать λ_t арифметического среднего для матрицы (симметрия, SPD), а upstream-λ применять только в переносе насыщенности (стандартная линеаризация IMPES).

```js
function materialBalance(sim) {
  let withdrawn = sim.cumProdM3, expansion = 0;
  for (const c of sim.r.cells) expansion += sim.r.poreVol[c] * sim.r.ct * (sim.P[c] - sim.r.P0);
  return { withdrawn, expansion, err: (withdrawn - expansion) / Math.max(1, withdrawn) };
}
```

```js
function createSim(reservoir, params) {
  const r = reservoir;
  const sim = {
    r, params, P: new Float64Array(r.nx * r.ny).fill(r.P0),
    Sw: new Float64Array(r.nx * r.ny).fill(r.Swc),
    wells: [], t: 0, cumProdM3: 0, lastCG: { iters: 0, resid: 0, rhs0: 1 },
    _rowOf: new Int32Array(r.nx * r.ny).fill(-1), // cell → строка матрицы
    _Pnew: null, _Ap: null, _res: null, _pz: null, // рабочие
  };
  r.cells.forEach((c, idx) => (sim._rowOf[c] = idx));
  const N = r.cells.length;
  sim._Pnew = new Float64Array(N); sim._Ap = new Float64Array(N);
  sim._res = new Float64Array(N); sim._pz = new Float64Array(N);
  sim.addWell = (c, type, pwf) => { sim.wells.push({ cell: c, type, pwf, color: null, name: null,
    q: 0, qo: 0, qw: 0, cum: { oil: 0, water: 0, inj: 0 } }); };
  sim.avgPressure = () => { let s = 0, v = 0; for (const c of r.cells) { s += r.poreVol[c] * sim.P[c]; v += r.poreVol[c]; } return s / v; };
  return sim;
}
const NB = (c, r) => [c % r.nx > 0 ? c - 1 : -1, c % r.nx < r.nx - 1 ? c + 1 : -1,
                      c - r.nx, c + r.nx]; // [запад, восток, юг(j-1), север(j+1)]
function faceT(sim, c, nb) { // симметричная проводимость с λ арифм. среднего
  const r = sim.r;
  const kH = 2 * r.k[c] * r.k[nb] / (r.k[c] + r.k[nb]); // гармоническое k
  const lam = 0.5 * (lambdaT(sim.Sw[c], r) + lambdaT(sim.Sw[nb], r));
  return CONV_K * kH * r.h * (r.dx / r.dx) * lam; // м³/сут/атм
}
function solvePressure(sim, dt) {
  const r = sim.r, N = r.cells.length;
  const { _Pnew: x, _Ap: Ap, _res: res, _pz: p } = sim;
  const acc = i => r.poreVol[i] * r.ct / dt;
  // правая часть и диагонали
  const diag = new Float64Array(N), b = new Float64Array(N);
  const Pold = sim.P;
  for (let idx = 0; idx < N; idx++) {
    const c = r.cells[idx];
    diag[idx] += acc(c);
    b[idx] = acc(c) * Pold[c];
    for (const nb of NB(c, r)) {
      if (nb < 0 || !r.active[nb]) continue;
      const T = faceT(sim, c, nb), j = sim._rowOf[nb];
      diag[idx] += T; b[idx] += -0; // поток учитывается в Ap ниже
      // Ap: -T·P_nb; b: ничего (поток линеен по P)
      b[idx] -= 0; // (оставлено для симметрии кода)
    }
  }
  // скважины (неявно): diag += WI·λt(cell); b += WI·λt·Pwf
  for (const w of sim.wells) {
    const idx = sim._rowOf[w.cell], lam = lambdaT(sim.Sw[w.cell], r);
    const G = wellIndex(r.k[w.cell], r) * lam;
    diag[idx] += G; b[idx] += G * w.pwf;
  }
  // CG с Якоби-предобуславливателем, warm start
  for (let idx = 0; idx < N; idx++) x[idx] = Pold[r.cells[idx]];
  const applyA = (v, out) => {
    for (let idx = 0; idx < N; idx++) {
      const c = r.cells[idx]; let s = diag[idx] * v[idx];
      for (const nb of NB(c, r)) {
        if (nb < 0 || !r.active[nb]) continue;
        s -= faceT(sim, c, nb) * v[sim._rowOf[nb]];
      }
      out[idx] = s;
    }
  };
  applyA(x, Ap);
  let rhs0 = 0; for (let idx = 0; idx < N; idx++) { res[idx] = b[idx] - Ap[idx]; rhs0 += b[idx] * b[idx]; }
  rhs0 = Math.sqrt(rhs0);
  const z = sim._pz;
  let rk = 0; for (let idx = 0; idx < N; idx++) { z[idx] = res[idx] / diag[idx]; rk += res[idx] * res[idx]; }
  let pk = z.slice(), rz = 0; for (let idx = 0; idx < N; idx++) rz += res[idx] * z[idx];
  let iters = 0, resid = Math.sqrt(rk);
  while (resid > 1e-6 * rhs0 && iters < 500) {
    applyA(pk, Ap);
    let pAp = 0; for (let idx = 0; idx < N; idx++) pAp += pk[idx] * Ap[idx];
    const ak = rz / pAp;
    for (let idx = 0; idx < N; idx++) { x[idx] += ak * pk[idx]; res[idx] -= ak * Ap[idx]; }
    for (let idx = 0; idx < N; idx++) z[idx] = res[idx] / diag[idx];
    let rzNew = 0, r2 = 0; for (let idx = 0; idx < N; idx++) { rzNew += res[idx] * z[idx]; r2 += res[idx] * res[idx]; }
    const bk = rzNew / rz; rz = rzNew; resid = Math.sqrt(r2);
    for (let idx = 0; idx < N; idx++) pk[idx] = z[idx] + bk * pk[idx];
    iters++;
  }
  sim.lastCG = { iters, resid, rhs0 };
  for (let idx = 0; idx < N; idx++) sim.P[r.cells[idx]] = x[idx];
}
```

```js
function computeFaceFluxes(sim) { // u[c*4+d]: >0 — из c в соседа d (запад/восток/юг/север)

function transportSubstep(sim, dt, u) {
  const r = sim.r, dSw = new Float64Array(r.nx * r.ny);
  for (const c of r.cells) {
    const dirs = NB(c, r); let net = 0;
    for (let d = 0; d < 4; d++) {
      const uf = u[c * 4 + d];
      if (uf > 0) net -= uf * fw(sim.Sw[c], r);          // вытекает с fw ячейки
      else { const nb = dirs[d];
        if (nb >= 0 && r.active[nb]) net += (-uf) * fw(sim.Sw[nb], r); } // втекает с fw соседа
    }
    for (const w of sim.wellAt?.[c] ?? []) { /* максимум 1 скважина на ячейку: см. addWell */ }
    dSw[c] = net / r.poreVol[c] * dt;
  }
  for (const w of sim.wells) {
    if (w.type === "inj") dSw[w.cell] += w.q * dt / r.poreVol[w.cell];
    else dSw[w.cell] -= w.qw * dt / r.poreVol[w.cell];
  }
  for (const c of r.cells) sim.Sw[c] = Math.min(1 - r.Sor, Math.max(r.Swc, sim.Sw[c] + dSw[c]));
}
function cflLimit(sim, u) {
  const r = sim.r; let dt = 2;
  for (const c of r.cells) {
    const dirs = NB(c, r); let inflow = 0;
    for (let d = 0; d < 4; d++) { const uf = u[c * 4 + d]; if (uf < 0) inflow -= uf; }
```
